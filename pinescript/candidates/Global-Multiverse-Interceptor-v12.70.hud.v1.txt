ext_drawCurZone(bool update) =>
    var line ext_pocLine = na
    var line ext_vahLine = na
    var line ext_valLine = na
    var box[] ext_redBoxes = array.new_box(ext_resolution, na)
    var box[] ext_greenBoxes = array.new_box(ext_resolution, na)

    // Clear disabled visual layers immediately; calculations/profile state remain intact.
    if barstate.islast
        if not ext_showProf or use_zen_mode
            for i = 0 to array.size(ext_redBoxes) - 1
                box rb = array.get(ext_redBoxes, i)
                box gb = array.get(ext_greenBoxes, i)
                if not na(rb)
                    box.delete(rb)
                    array.set(ext_redBoxes, i, na)
                if not na(gb)
                    box.delete(gb)
                    array.set(ext_greenBoxes, i, na)
        if not ext_showPoc and not na(ext_pocLine)
            line.delete(ext_pocLine)
            ext_pocLine := na
        if not ext_showVA
            if not na(ext_vahLine)
                line.delete(ext_vahLine)
                ext_vahLine := na
            if not na(ext_valLine)
                line.delete(ext_valLine)
                ext_valLine := na

    if update and ext_validProfileRange and array.sum(ext_vpGreen) + array.sum(ext_vpRed) > 0
        if not na(ext_pocLine)
            line.delete(ext_pocLine)
        if not na(ext_vahLine)
            line.delete(ext_vahLine)
        if not na(ext_valLine)
            line.delete(ext_valLine)

        for i = 0 to array.size(ext_redBoxes) - 1
            if not na(array.get(ext_redBoxes, i))
                box.delete(array.get(ext_redBoxes, i))
                box.delete(array.get(ext_greenBoxes, i))
        
        float gap = (ext_profHigh - ext_profLow) / ext_resolution
        // Compute historical bar indices arithmetically to avoid dynamic series offsets
        // exceeding Pine's inferred historical buffer (e.g. bar_index[37] with buffer 36).
        float leftMax = bar_index - ext_lookback
        float rightMax = bar_index - int(ext_lookback / 1.4)
        float rightMaxVol = array.max(ext_vpGreen)+array.max(ext_vpRed)
        float buffer = gap / 10
        
        if ext_showProf and not use_zen_mode
            for i = 0 to array.size(ext_vpRed) - 1
                int greenEnd = int(leftMax + (rightMax - leftMax) * (array.get(ext_vpGreen, i) / math.max(rightMaxVol, 0.0001)))
                int redEnd = int(greenEnd + (rightMax - leftMax) * (array.get(ext_vpRed, i) / math.max(rightMaxVol, 0.0001)))
                
                array.set(ext_greenBoxes, i, box.new(int(leftMax), array.get(ext_zoneBounds, i) - buffer, greenEnd, array.get(ext_zoneBounds, i) - gap + buffer, bgcolor=ext_bullCol, border_width=0))
                array.set(ext_redBoxes, i, box.new(greenEnd, array.get(ext_zoneBounds, i) - buffer, redEnd, array.get(ext_zoneBounds, i) - gap + buffer, bgcolor=ext_bearCol, border_width=0))
        
        float cur_poc = lib.ext_profile_poc(ext_validProfileRange, ext_vpRed, ext_vpGreen, ext_zoneBounds)
        [cur_val, cur_vah] = lib.ext_profile_value_levels(cur_poc, ext_validProfileRange, ext_profHigh, ext_profLow, ext_resolution, ext_VAwid, ext_vpRed, ext_vpGreen, ext_zoneBounds)
        
        if ext_showPoc and not na(cur_poc)
            ext_pocLine := line.new(int(leftMax), cur_poc, bar_index, cur_poc, color=ext_pocCol, width=2)
        if ext_showVA and not na(cur_vah) and not na(cur_val)
            ext_vahLine := line.new(int(leftMax), cur_vah, bar_index, cur_vah, color=ext_vahCol, width=1, style=line.style_dashed)
            ext_valLine := line.new(int(leftMax), cur_val, bar_index, cur_val, color=ext_vahCol, width=1, style=line.style_dashed)
    else if update
        // Remove stale levels/profile when a new session has no valid range or volume.
        if not na(ext_pocLine)
            line.delete(ext_pocLine)
            ext_pocLine := na
        if not na(ext_vahLine)
            line.delete(ext_vahLine)
            ext_vahLine := na
        if not na(ext_valLine)
            line.delete(ext_valLine)
            ext_valLine := na
        for i = 0 to array.size(ext_redBoxes) - 1
            if not na(array.get(ext_redBoxes, i))
                box.delete(array.get(ext_redBoxes, i))
                array.set(ext_redBoxes, i, na)
            if not na(array.get(ext_greenBoxes, i))
                box.delete(array.get(ext_greenBoxes, i))
                array.set(ext_greenBoxes, i, na)

    // Explicit, consistent return type for both drawing and cleanup branches.
    0

// Session identity is computed from each intrabar timestamp, not just chart bars.
// Regional sessions use IANA timezones so London/New York DST is handled by Pine.
ext_calendarTimezone = ext_calendar_tz_mode == 'Exchange' ? syminfo.timezone : 'Etc/UTC'
var int ext_activeSessionKey = na
[ext_sessionReset, ext_nextSessionKey] = lib.ext_update_intra(e_dT, e_dO, e_dH, e_dL, e_dC, e_dV, ext_ltfTime, ext_ltfOpen, ext_ltfHigh, ext_ltfLow, ext_ltfClose, ext_ltfVolume, ext_activeSessionKey, ext_calendarTimezone, ext_sessionType, ext_tokyo_start_h, ext_tokyo_start_m, ext_tokyo_end_h, ext_tokyo_end_m, ext_london_start_h, ext_london_start_m, ext_london_end_h, ext_london_end_m, ext_ny_start_h, ext_ny_start_m, ext_ny_end_h, ext_ny_end_m)
ext_activeSessionKey := ext_nextSessionKey
if ext_sessionReset
    ext_zoneStart := bar_index
ext_lookback := bar_index - ext_zoneStart
// Re-evaluate after ingest because arrays may have been reset/populated above.
ext_profHigh := array.size(ext_ltfHigh) > 0 ? array.max(ext_ltfHigh) : na
ext_profLow := array.size(ext_ltfLow) > 0 ? array.min(ext_ltfLow) : na
ext_validProfileRange := not na(ext_profHigh) and not na(ext_profLow) and ext_profHigh > ext_profLow and ext_resolution >= 2
lib.ext_profile_calc(ext_validProfileRange, barstate.islast, ext_profHigh, ext_profLow, ext_resolution, ext_ltfOpen, ext_ltfHigh, ext_ltfLow, ext_ltfClose, ext_ltfVolume, ext_zoneBounds, ext_vpGreen, ext_vpRed)
ext_drawCurZone(barstate.islast)
