export kww_calc(series float src, simple int len) =>
    float tau  = TAU()
    float beta = BETA()
    float sum_w  = 0.0
    float sum_pw = 0.0
    for i = 0 to len - 1
        float w = math.exp(-math.pow(i / tau, beta))
        sum_w  += w
        sum_pw += nz(src[i], src) * w
    float ma_val = sum_pw / math.max(sum_w, 0.0001)
   
    float sum_dev = 0.0
    for i = 0 to len - 1
        float w = math.exp(-math.pow(i / tau, beta))
        sum_dev += w * math.pow(nz(src[i], src) - ma_val, 2)
    float sig_val = math.sqrt(sum_dev / math.max(sum_w, 0.0001))
    [ma_val, sig_val]

export kww_zscore(series float src, simple int len) =>
    [m, s] = kww_calc(src, len)
    (src - m) / math.max(s, 0.0001)

export hurst_rs(series float src, simple int len) =>
    if bar_index < len
        0.5
    else
        float mean_ret = 0.0
        for i = 0 to len - 1
            mean_ret += math.log(src[i] / nz(src[i+1], src[i]))
        mean_ret /= len
        float max_dev = 0.0
        float min_dev = 0.0
        float cum_dev = 0.0
    
        float var_ret = 0.0
        for i = 0 to len - 1
            float r = math.log(src[i] / nz(src[i+1], src[i]))
            cum_dev += (r - mean_ret)
            max_dev := math.max(max_dev, cum_dev)
            min_dev := math.min(min_dev, cum_dev)
            var_ret += math.pow(r - mean_ret, 2)
        float R = max_dev - min_dev
        float S = math.sqrt(var_ret / (len - 1))
        float H = (S == 0.0 or R == 0.0) ? 0.5 : math.log(R / S) / math.log(len)
        math.max(0.0, math.min(1.0, H))

export q_gravity(array<float> bear_targets, array<float> bear_mass, array<float> bull_targets, array<float> bull_mass, float atr_val) =>
    float q_f_up = 0.0
    float q_f_dn = 0.0
    if array.size(bear_targets) > 0
        for i = 0 to array.size(bear_targets) - 1
            float tgt  = array.get(bear_targets, i)
            float mass = nz(array.get(bear_mass, i), 0.0)
            if tgt > close
                float r_dist = math.max((tgt - close) / math.max(atr_val, 0.0001), 0.1)
                q_f_up += mass / math.pow(r_dist, 2)
    if array.size(bull_targets) > 0
        for i = 0 to array.size(bull_targets) - 1
            float tgt  = array.get(bull_targets, i)
            float mass = nz(array.get(bull_mass, i), 0.0)
            if tgt < close
                float r_dist = math.max((close - tgt) / math.max(atr_val, 0.0001), 0.1)
                q_f_dn += mass / math.pow(r_dist, 2)
    float q_total       = q_f_up + q_f_dn
    float q_net_gravity = q_total > 0 ? (q_f_up - q_f_dn) / q_total : 0.0
    float q_up_pct      = q_total > 0 ? (q_f_up / q_total) * 100.0 : 50.0
    float q_dn_pct      = q_total > 0 ? (q_f_dn / q_total) * 100.0 : 50.0
    float raw_g_scale   = q_net_gravity * 0.618
    float g_scale       = nz(ta.ema(raw_g_scale, 5), raw_g_scale)
    [q_f_up, q_f_dn, q_net_gravity, q_up_pct, q_dn_pct, g_scale]

export psi_adaptive(float psi_f, float psi_m, float psi_s, float psi_d,
                    float gamma_f, float gamma_m, float gamma_s, float gamma_d, float eff_ratio) =>
    float eff     = math.max(0.1, eff_ratio)
    float p_inv_eff = math.max(0.1, 1.0 - eff_ratio)
    float aif_f   = math.abs(psi_f)
    float aif_m   = math.abs(psi_m)
    float aif_s   = math.abs(psi_s)
    float aif_d   = math.abs(psi_d)
    float total_w = (aif_f * eff) + (aif_m * 1.0) + (aif_s * p_inv_eff) + (aif_d * 0.5 * p_inv_eff)
    float w_f = total_w == 0 ? 0.25 : (aif_f * eff) / total_w
    float w_m = total_w == 0 ? 0.25 : (aif_m * 1.0) / total_w
    float w_s = total_w == 0 ? 0.25 : (aif_s * p_inv_eff) / total_w
    float w_d = total_w == 0 ? 0.25 : (aif_d * 0.5 * p_inv_eff) / total_w
    float raw = (psi_f * gamma_f * w_f) + (psi_m * gamma_m * w_m) + (psi_s * gamma_s * w_s) + (psi_d * gamma_d * w_d)
    math.max(-1.0, math.min(raw, 1.0))

    [tensor_fast, tensor_slow, tensor_core]

export get_jurik(simple int length, simple int phase, simple int power, series float src) =>
    float phaseRatio = phase < -100 ? 0.5 : phase > 100 ? 2.5 : phase / 100 + 1.5
    float beta = 0.45 * (length - 1) / (0.45 * (length - 1) + 2)
    float alpha = math.pow(beta, power)
    float jma = 0.0
    float e0 = 0.0
    float e1 = 0.0
    float e2 = 0.0
    e0 := (1 - alpha) * src + alpha * nz(e0[1])
    e1 := (src - e0) * (1 - beta) + beta * nz(e1[1])
    e2 := (e0 + phaseRatio * e1 - nz(jma[1])) * math.pow(1 - alpha, 2) + math.pow(alpha, 2) * nz(e2[1])
    jma := e2 + nz(jma[1])
    jma

export ob_zone_bounds(series float src_high, series float src_low, series float src_atr, simple int pivot_len, float max_span_atr, int direction) =>
    float source_high = src_high[pivot_len]
    float source_low = src_low[pivot_len]

    float source_atr = math.max(nz(src_atr[pivot_len], ta.atr(14)[pivot_len]), syminfo.mintick)
    float span_cap = math.max(source_atr * max_span_atr, syminfo.mintick)

    float bottom = direction == -1 ? math.max(source_mid, source_high - span_cap) : source_low
    float midpoint = math.avg(top, bottom)
    float initial_span = math.max(top - bottom, syminfo.mintick)
    [top, bottom, midpoint, initial_span]

// Shared progressive mitigation. Returns updated top/bottom, remaining %, touch, invalid.
// Wick penetration carves the consumed edge; close_only controls full-break invalidation.
export ob_progressive_update(int direction, float zone_top, float zone_bottom, float initial_span, float atr_locked, series float src_high, series float src_low, series float src_close, float buffer_atr, bool allow_update, bool close_only) =>
    float next_top = zone_top
    float next_bottom = zone_bottom
    float buffer = math.max(nz(atr_locked, 0.0) * buffer_atr, 0.0)
    bool touched = allow_update and src_high >= zone_bottom and src_low <= zone_top
    if allow_update
        if direction == -1 and src_high > zone_bottom and src_low <= zone_top
            next_bottom := math.min(zone_top, math.max(zone_bottom, src_high))
        else if direction == 1 and src_low < zone_top and src_high >= zone_bottom
            next_top := math.max(zone_bottom, math.min(zone_top, src_low))
    bool consumed = (next_top - next_bottom) <= syminfo.mintick
    bool broken_wick = direction == -1 ? src_high >= next_top : src_low <= next_bottom
    bool broken_close = direction == -1 ? src_close > next_top + buffer : src_close < next_bottom - buffer
    bool invalid = allow_update and (consumed or (close_only ? broken_close : (broken_wick or broken_close)))
    float remaining = math.round(((next_top - next_bottom) / math.max(initial_span, syminfo.mintick)) * 100.0)
    [next_top, next_bottom, remaining, touched, invalid]

export ob_retest_count(series float src_high, series float src_low, float zone_top, float zone_bottom, float touch_band, simple int lookback) =>

            if back - last_touch_back >= 3
                touches += 1
                last_touch_back := back
    touches

    distance_score + age_score + retest_score + width_score + structure_score + (cycle_aligned ? 5.0 : 0.0) + (macro_aligned ? 5.0 : 0.0) + (prophet_gmi_aligned ? 5.0 : 0.0)

// Shared CSHI spread/velocity viability calculations.
export cshi_engine(series float src_open, series float src_high, series float src_low, series float src_close, series float atr_value, series float efficiency, series float entropy) =>
    float body = math.abs(src_close - src_open)
    float range_value = src_high - src_low
    float wick = range_value - body
    float spread_toxicity = wick > 0 ? body / math.max(range_value, 0.0001) : 1.0
    float smooth_spread = ta.ema(spread_toxicity, 5)
    float price_velocity = math.abs(src_close - nz(src_close[1], src_close)) / math.max(atr_value, 0.0001)
    float smooth_velocity = ta.ema(price_velocity, 5)
    float viability_score = smooth_spread * 0.4 + smooth_velocity * 0.4 + (1.0 - efficiency) * 0.2
    float grid = nz(ta.percentrank(viability_score, 100), 50.0)
    bool muddy = entropy >= 0.80 and grid >= 45.0 and grid <= 85.0
    bool viable = muddy or (grid >= 50.0 and grid <= 85.0)
    [spread_toxicity, smooth_spread, price_velocity, smooth_velocity, viability_score, grid, muddy, viable]

// Unified TDM volume profile + KDE-style smoothing + value area + void scanner.
// Return order: min, max, step, raw/smoothed bins, entropy bins, counts,
// VAH, VAL, POC, bull LVN1/2, bear LVN1/2, bull/bear HVN, gravity up/down/net.
export tdm_profile(series float src_high, series float src_low, series float src_volume, series float src_hlc3, series float entropy, series float src_close, series float atr_value, simple int lookback, simple int rows) =>
    float global_min = ta.lowest(src_low, lookback)
    float global_max = ta.highest(src_high, lookback)
    float bin_step = (global_max - global_min) == 0 ? syminfo.mintick : (global_max - global_min) / rows
    array<float> bins = array.new_float(rows, 0.0)
    array<float> entropy_bins = array.new_float(rows, 0.0)
    array<int> count_bins = array.new_int(rows, 0)
    for i = 0 to lookback - 1
        if not na(src_low[i]) and not na(src_high[i]) and not na(src_hlc3[i])
            float bar_volume = math.max(nz(src_volume[i], 0.0), 0.0)
            int idx_low = math.max(0, math.min(rows - 1, int(math.floor((src_low[i] - global_min) / bin_step))))
            int idx_mid = math.max(0, math.min(rows - 1, int(math.floor((src_hlc3[i] - global_min) / bin_step))))
            int idx_high = math.max(0, math.min(rows - 1, int(math.floor((src_high[i] - global_min) / bin_step))))
            if idx_low == idx_mid and idx_mid == idx_high
                array.set(bins, idx_mid, array.get(bins, idx_mid) + bar_volume)
            else
                array.set(bins, idx_low, array.get(bins, idx_low) + bar_volume * 0.25)
                array.set(bins, idx_mid, array.get(bins, idx_mid) + bar_volume * 0.50)
                array.set(bins, idx_high, array.get(bins, idx_high) + bar_volume * 0.25)
            array.set(entropy_bins, idx_mid, array.get(entropy_bins, idx_mid) + nz(entropy[i], 0.5))
            array.set(count_bins, idx_mid, array.get(count_bins, idx_mid) + 1)
    array<float> smooth_bins = array.new_float(rows, 0.0)
    for i = 0 to rows - 1
        float vm2 = i >= 2 ? array.get(bins, i - 2) : 0.0
        float vm1 = i >= 1 ? array.get(bins, i - 1) : 0.0
        float v0 = array.get(bins, i)
        float vp1 = i <= rows - 2 ? array.get(bins, i + 1) : 0.0
        float vp2 = i <= rows - 3 ? array.get(bins, i + 2) : 0.0
        array.set(smooth_bins, i, (vm2 + vm1 * 2.0 + v0 * 3.0 + vp1 * 2.0 + vp2) / 9.0)
    float max_volume = 0.0
    float total_volume = 0.0
    int poc_index = 0
    for i = 0 to rows - 1
        float sv = array.get(smooth_bins, i)
        array.set(bins, i, sv)
        if sv > max_volume
            max_volume := sv
            poc_index := i
        total_volume += sv
    float target_value_area = total_volume * (0.65 + nz(entropy, 0.5) * 0.15)
    float value_area_volume = max_volume
    int up_index = poc_index + 1
    int down_index = poc_index - 1
    while value_area_volume < target_value_area and (up_index < rows or down_index >= 0)
        float up_volume = up_index < rows ? array.get(bins, up_index) : -1.0
        float down_volume = down_index >= 0 ? array.get(bins, down_index) : -1.0
        if up_volume >= down_volume and up_volume != -1.0
            value_area_volume += up_volume
            up_index += 1
        else if down_volume != -1.0
            value_area_volume += down_volume
            down_index -= 1
    float vah = global_min + math.min(rows, up_index) * bin_step
    float val = global_min + math.max(0, down_index + 1) * bin_step
    float poc = global_min + poc_index * bin_step + bin_step * 0.5
    float lvn_bull_1 = na
    float lvn_bull_2 = na
    float lvn_bear_1 = na
    float lvn_bear_2 = na
    float hvn_bull = na
    float hvn_bear = na
    float gravity_up = 0.0
    float gravity_down = 0.0
    for i = 1 to rows - 2
        float v = array.get(bins, i)
        float price = global_min + i * bin_step
        float center = price + bin_step * 0.5
        if v > 0
            float distance = math.max(math.abs(src_close - center) / atr_value, 0.1)
            float pull = v / math.pow(distance, 2)
            if center > src_close
                gravity_up += pull
            else if center < src_close
                gravity_down += pull
        if v < array.get(bins, i - 1) and v < array.get(bins, i + 1) and v < max_volume * 0.4
            if center <= src_close + atr_value * 0.5
                lvn_bull_2 := lvn_bull_1
                lvn_bull_1 := center
            if center >= src_close - atr_value * 0.5 and center != lvn_bull_1
                if na(lvn_bear_1)
                    lvn_bear_1 := center
                else if na(lvn_bear_2)
                    lvn_bear_2 := center
        if v > array.get(bins, i - 1) and v > array.get(bins, i + 1) and v > max_volume * 0.6
            if price < src_close
                hvn_bull := price
            if price > src_close
                hvn_bear := price
    float gravity_total = gravity_up + gravity_down
    float gravity_net = gravity_total > 0 ? (gravity_up - gravity_down) / gravity_total * 100.0 : 0.0
    [global_min, global_max, bin_step, bins, smooth_bins, entropy_bins, count_bins, vah, val, poc, lvn_bull_1, lvn_bull_2, lvn_bear_1, lvn_bear_2, hvn_bull, hvn_bear, gravity_up, gravity_down, gravity_net]

// Full Prophet R37 divergence/structure state machine. Rendering stays in Prophet.
// subtype: 0 unresolved, 1 bull trap, 2 true breakout, 3 bear trap, 4 true breakdown.
export prophet_structure_core(series float oscillator, series float src_high, series float src_low, series float src_open, series float src_close, series float volume_ratio, series bool has_volume, series float atr_value, bool legacy_mode, simple int state_max_age, simple int pivot_left, simple int pivot_right, float body_floor_atr, float wick_body_ratio) =>
    var float last_ph_value = na
    var float last_price_ph = na
    var int last_ph_index = na
    var float last_pl_value = na
    var float last_price_pl = na
    var int last_pl_index = na
    var float d_state = 0.0
    var float b_state = 0.0
    var int state_bar = na
    var int bull_event_bar = na
    var int bear_event_bar = na
    var int subtype = 0
    float oscillator_ph = ta.pivothigh(oscillator, pivot_left, pivot_right)
    float oscillator_pl = ta.pivotlow(oscillator, pivot_left, pivot_right)
    bool cross_up = ta.crossover(oscillator, 0.0)
    bool cross_down = ta.crossunder(oscillator, 0.0)
    bool bear_div_event = false
    bool bull_div_event = false
    int bear_div_origin_bar = na
    float bear_div_origin_value = na
    int bull_div_origin_bar = na
    float bull_div_origin_value = na
    if (cross_up or cross_down) and (math.abs(d_state) < 3.0 or not legacy_mode)
        d_state := 0.0
        b_state := 1.0
        if not legacy_mode
            subtype := 0
            state_bar := na
            bull_event_bar := na
            bear_event_bar := na
    if not na(oscillator_ph)
        if src_high[pivot_right] > last_price_ph and oscillator_ph < last_ph_value and oscillator_ph > 0
            d_state := -1.0
            b_state := 2.0
            bear_event_bar := bar_index
            bear_div_event := true
            bear_div_origin_bar := last_ph_index
            bear_div_origin_value := last_ph_value
            if not legacy_mode
                bull_event_bar := na
        last_ph_value := oscillator_ph
        last_price_ph := src_high[pivot_right]
        last_ph_index := bar_index - pivot_right
    if not na(oscillator_pl)
        if src_low[pivot_right] < last_price_pl and oscillator_pl > last_pl_value and oscillator_pl < 0
            d_state := 1.0
            b_state := 2.0
            bull_event_bar := bar_index
            bull_div_event := true
            bull_div_origin_bar := last_pl_index
            bull_div_origin_value := last_pl_value
            if not legacy_mode
                bear_event_bar := na
        last_pl_value := oscillator_pl
        last_price_pl := src_low[pivot_right]
        last_pl_index := bar_index - pivot_right
    float body_value = math.abs(src_close - src_open)
    float lower_wick = math.min(src_open, src_close) - src_low
    float upper_wick = src_high - math.max(src_open, src_close)
    float effective_body = legacy_mode ? body_value : math.max(body_value, atr_value * body_floor_atr)
    bool bottom_rejection = lower_wick > effective_body * wick_body_ratio
    bool top_rejection = upper_wick > effective_body * wick_body_ratio
    bool real_volume = legacy_mode ? volume_ratio > 1.0 : (has_volume and volume_ratio > 1.0)
    if d_state == -1.0
        if src_close > last_price_ph
            if top_rejection or (legacy_mode ? not real_volume : (has_volume and not real_volume))
                d_state := 3.0
                b_state := -4.0
                subtype := 1
                bear_event_bar := bar_index
                if not legacy_mode
                    bull_event_bar := na
            else
                d_state := -3.0
                b_state := 3.0
                subtype := 2
                bull_event_bar := bar_index
                if not legacy_mode
                    bear_event_bar := na
            state_bar := bar_index
        else if src_close < src_low[1] and src_close < src_open
            b_state := -3.0
            bear_event_bar := bar_index
            if not legacy_mode
                bull_event_bar := na
    if d_state == 1.0
        if src_close < last_price_pl
            if bottom_rejection or (legacy_mode ? not real_volume : (has_volume and not real_volume))
                d_state := -3.0
                b_state := 3.0
                subtype := 3
                bull_event_bar := bar_index
                if not legacy_mode
                    bear_event_bar := na
            else
                d_state := 3.0
                b_state := -4.0
                subtype := 4
                bear_event_bar := bar_index
                if not legacy_mode
                    bull_event_bar := na
            state_bar := bar_index
        else if src_close > src_high[1] and src_close > src_open
            b_state := 3.0
            bull_event_bar := bar_index
            if not legacy_mode
                bear_event_bar := na
    if not legacy_mode and (d_state == 3.0 or d_state == -3.0) and not na(state_bar) and bar_index - state_bar > state_max_age
        d_state := 0.0
        b_state := 1.0
        subtype := 0
        state_bar := na
        bull_event_bar := na
        bear_event_bar := na
    [d_state, b_state, state_bar, bull_event_bar, bear_event_bar, subtype, oscillator_ph, oscillator_pl, bear_div_event, bull_div_event, bear_div_origin_bar, bear_div_origin_value, bull_div_origin_bar, bull_div_origin_value, last_price_ph, last_price_pl]

export camarilla_extended(series float src_high, series float src_low, series float src_close) =>
    float ph = nz(src_high[1], src_high)
    float pl = nz(src_low[1], src_low)
    float pc = nz(src_close[1], src_close)
    float price_range = math.max(ph - pl, 0.0001)
    float h4 = pc + price_range * 1.1 / 2.0
    float h3 = pc + price_range * 1.1 / 4.0
    float l3 = pc - price_range * 1.1 / 4.0
    float l4 = pc - price_range * 1.1 / 2.0
    float h5 = h4 + 1.168 * (h4 - h3)
    float l5 = l4 - 1.168 * (l3 - l4)
    float h6 = h5 + 1.168 * (h5 - h4)
    float l6 = l5 - 1.168 * (l4 - l5)
    [h6, h5, h4, h3, l3, l4, l5, l6]

export daily_cpr_stats(series float src_high, series float src_low, series float src_close) =>
    float previous_pivot = (src_high[1] + src_low[1] + src_close[1]) / 3.0
    float previous_bc = (src_high[1] + src_low[1]) / 2.0
    float previous_tc = 2.0 * previous_pivot - previous_bc
    float previous_width = math.abs(previous_tc - previous_bc)
    float average_width_20 = ta.sma(previous_width, 20)
    [src_high[1], src_low[1], src_close[1], average_width_20]

export macro_kinematics(series float src_close) =>
    float roc_value = ta.roc(src_close, 14)
    float regression_now = ta.linreg(roc_value, 14, 0)
    float regression_previous = ta.linreg(roc_value, 14, 1)
    float angle = math.atan((regression_now - regression_previous) * 10.0) * (180.0 / math.pi)
    float angular_velocity = ta.change(angle)
    [ta.ema(angle, 3), ta.ema(angular_velocity, 3)]

export return_distribution(series float src_close, series float moving_average, series float sigma, series float entropy) =>
    float returns = (src_close - nz(src_close[1], src_close)) / math.max(nz(src_close[1], src_close), 0.0001)
    float base_mean = ta.sma(returns, 50)
    float adjusted_mean = base_mean + ((src_close - moving_average) / math.max(sigma, 0.0001) * ta.stdev(returns, 50) * 0.1)
    float adjusted_sigma = ta.stdev(returns, 50) * (1.0 + entropy)
    [returns, base_mean, adjusted_mean, adjusted_sigma]

export gabfx_update(series float src_close, series float recent_pivot_high, series float previous_pivot_high, series float recent_pivot_low, series float previous_pivot_low, float prior_qm_sell, float prior_qm_buy, float prior_rbs, float prior_sbr, float atr_value) =>
    float qm_sell = prior_qm_sell
    float qm_buy = prior_qm_buy
    float rbs = prior_rbs
    float sbr = prior_sbr
    bool break_support = ta.crossunder(src_close, recent_pivot_low)
    bool break_resistance = ta.crossover(src_close, recent_pivot_high)
    bool higher_high = recent_pivot_high > previous_pivot_high
    bool higher_low = recent_pivot_low > previous_pivot_low
    bool lower_low = recent_pivot_low < previous_pivot_low
    bool lower_high = recent_pivot_high < previous_pivot_high
    if higher_high and higher_low and break_support
        qm_sell := previous_pivot_high
    if lower_low and lower_high and break_resistance
        qm_buy := previous_pivot_low
    if break_resistance
        rbs := recent_pivot_high
    if break_support
        sbr := recent_pivot_low
    if not na(qm_buy) and src_close < qm_buy
        qm_buy := na
    if not na(qm_sell) and src_close > qm_sell
        qm_sell := na
    if not na(rbs) and src_close < rbs
        rbs := na
    if not na(sbr) and src_close > sbr
        sbr := na
    bool in_qm_buy = not na(qm_buy) and math.abs(src_close - qm_buy) <= atr_value
    bool in_qm_sell = not na(qm_sell) and math.abs(src_close - qm_sell) <= atr_value
    bool in_rbs = not na(rbs) and math.abs(src_close - rbs) <= atr_value
    bool in_sbr = not na(sbr) and math.abs(src_close - sbr) <= atr_value
    [qm_sell, qm_buy, rbs, sbr, in_qm_buy, in_qm_sell, in_rbs, in_sbr]

export ext_volume_overlap(float y11, float y12, float y21, float y22, float height, float vol) =>
    nz(math.max(math.min(math.max(y11, y12), math.max(y21, y22)) - math.max(math.min(y11, y12), math.min(y21, y22)), 0.0) * vol / math.max(height, 0.0001))

export ext_profile_add(float o, float h, float l, float c, float v, float gap, array<float> zone_bounds, array<float> vp_green, array<float> vp_red) =>
    for i = 0 to array.size(vp_green) - 1
        float zone_top = array.get(zone_bounds, i)
        float zone_bot = zone_top - gap
        float body_top = math.max(c, o)
        float body_bot = math.min(c, o)
        bool is_green = c >= o
        float top_wick = h - body_top
        float bottom_wick = body_bot - l
        float body_size = body_top - body_bot
        float total_range = 2.0 * top_wick + 2.0 * bottom_wick + body_size
        float body_vol = total_range > 0 ? body_size * v / total_range : 0.0
        float top_wick_vol = total_range > 0 ? 2.0 * top_wick * v / total_range : 0.0
        float bottom_wick_vol = total_range > 0 ? 2.0 * bottom_wick * v / total_range : 0.0
        float green_add = (is_green ? ext_volume_overlap(zone_bot, zone_top, body_bot, body_top, body_size, body_vol) : 0.0) + ext_volume_overlap(zone_bot, zone_top, body_top, h, top_wick, top_wick_vol) / 2.0 + ext_volume_overlap(zone_bot, zone_top, body_bot, l, bottom_wick, bottom_wick_vol) / 2.0
        float red_add = (is_green ? 0.0 : ext_volume_overlap(zone_bot, zone_top, body_bot, body_top, body_size, body_vol)) + ext_volume_overlap(zone_bot, zone_top, body_top, h, top_wick, top_wick_vol) / 2.0 + ext_volume_overlap(zone_bot, zone_top, body_bot, l, bottom_wick, bottom_wick_vol) / 2.0
        array.set(vp_green, i, array.get(vp_green, i) + green_add)
        array.set(vp_red, i, array.get(vp_red, i) + red_add)
    0

export ext_profile_calc(bool valid_range, bool update, float prof_high, float prof_low, simple int resolution, array<float> ltf_open, array<float> ltf_high, array<float> ltf_low, array<float> ltf_close, array<float> ltf_volume, array<float> zone_bounds, array<float> vp_green, array<float> vp_red) =>
    array.fill(vp_green, 0.0)
    array.fill(vp_red, 0.0)
    if valid_range and update and array.size(ltf_open) > 0
        float gap = (prof_high - prof_low) / resolution
        if not na(gap) and gap > 0
            for i = 0 to resolution - 1
                array.set(zone_bounds, i, prof_high - gap * i)
            int n = math.min(array.size(ltf_open), math.min(array.size(ltf_high), math.min(array.size(ltf_low), math.min(array.size(ltf_close), array.size(ltf_volume)))))
            for j = 0 to n - 1
                ext_profile_add(array.get(ltf_open, j), array.get(ltf_high, j), array.get(ltf_low, j), array.get(ltf_close, j), array.get(ltf_volume, j), gap, zone_bounds, vp_green, vp_red)
    0

export ext_profile_poc(bool valid_range, array<float> vp_red, array<float> vp_green, array<float> zone_bounds) =>
    float max_volume = 0.0
    int level_index = 0
    int n_bins = array.size(vp_red)
    float out_level = na
    if valid_range and n_bins >= 2
        for i = 0 to n_bins - 1
            float bin_volume = array.get(vp_red, i) + array.get(vp_green, i)
            if bin_volume > max_volume
                max_volume := bin_volume
                level_index := i
        if max_volume > 0 and level_index < n_bins - 1
            out_level := array.get(zone_bounds, level_index) - (array.get(zone_bounds, level_index) - array.get(zone_bounds, level_index + 1)) / 2.0
    out_level

export ext_profile_value_levels(float ext_poc, bool valid_range, float prof_high, float prof_low, simple int resolution, float va_width, array<float> vp_red, array<float> vp_green, array<float> zone_bounds) =>
    float gap = valid_range ? (prof_high - prof_low) / resolution : na
    float volume_sum = array.sum(vp_red) + array.sum(vp_green)
    float volume_count = 0.0
    float ext_vah = na
    float ext_val = na
    int n_bins = array.size(vp_red)
    int poc_index = na
    if not na(ext_poc) and not na(gap) and gap > 0 and volume_sum > 0 and n_bins >= 2
        ext_vah := prof_high
        ext_val := prof_low
        for i = 0 to n_bins - 2
            if array.get(zone_bounds, i) >= ext_poc and array.get(zone_bounds, i + 1) < ext_poc
                poc_index := i
        if not na(poc_index)
            volume_count := array.get(vp_red, poc_index) + array.get(vp_green, poc_index)
            for i = 1 to n_bins
                if poc_index + i < n_bins
                    volume_count += array.get(vp_red, poc_index + i) + array.get(vp_green, poc_index + i)
                    if volume_count >= volume_sum * (va_width / 100.0)
                        break
                    else
                        ext_val := array.get(zone_bounds, poc_index + i) - gap
                if poc_index - i >= 0
                    volume_count += array.get(vp_red, poc_index - i) + array.get(vp_green, poc_index - i)
                    if volume_count >= volume_sum * (va_width / 100.0)
                        break
                    else
                        ext_vah := array.get(zone_bounds, poc_index - i)
    [ext_val, ext_vah]

export ext_session_key(int ts, simple string calendar_timezone, simple string session_type, simple int tokyo_start_h, simple int tokyo_start_m, simple int tokyo_end_h, simple int tokyo_end_m, simple int london_start_h, simple int london_start_m, simple int london_end_h, simple int london_end_m, simple int ny_start_h, simple int ny_start_m, simple int ny_end_h, simple int ny_end_m) =>
    int key = na
    int y = year(ts, calendar_timezone)
    int mo = month(ts, calendar_timezone)
    int d = dayofmonth(ts, calendar_timezone)
    if session_type == "Daily"
        key := y * 10000 + mo * 100 + d
    else if session_type == "Weekly"
        key := y * 100 + weekofyear(ts, calendar_timezone)
    else if session_type == "Monthly"
        key := y * 100 + mo
    else if session_type == "Quarterly"
        key := y * 10 + int(math.ceil(mo / 3.0))
    else if session_type == "Yearly"
        key := y
    else
        string session_timezone = session_type == "Tokyo" ? "Asia/Tokyo" : session_type == "London" ? "Europe/London" : "America/New_York"
        int local_minute = hour(ts, session_timezone) * 60 + minute(ts, session_timezone)
        int start_minute = session_type == "Tokyo" ? tokyo_start_h * 60 + tokyo_start_m : session_type == "London" ? london_start_h * 60 + london_start_m : ny_start_h * 60 + ny_start_m
        int end_minute = session_type == "Tokyo" ? tokyo_end_h * 60 + tokyo_end_m : session_type == "London" ? london_end_h * 60 + london_end_m : ny_end_h * 60 + ny_end_m
        bool in_session = start_minute < end_minute ? local_minute >= start_minute and local_minute < end_minute : start_minute > end_minute ? local_minute >= start_minute or local_minute < end_minute : false
        key := in_session ? y * 10000 + mo * 100 + d : na
    key

export ext_update_intra(array<int> intrabar_time, array<float> intrabar_open, array<float> intrabar_high, array<float> intrabar_low, array<float> intrabar_close, array<float> intrabar_volume, array<int> stored_time, array<float> stored_open, array<float> stored_high, array<float> stored_low, array<float> stored_close, array<float> stored_volume, int prior_active_key, simple string calendar_timezone, simple string session_type, simple int tokyo_start_h, simple int tokyo_start_m, simple int tokyo_end_h, simple int tokyo_end_m, simple int london_start_h, simple int london_start_m, simple int london_end_h, simple int london_end_m, simple int ny_start_h, simple int ny_start_m, simple int ny_end_h, simple int ny_end_m) =>
    int active_key = prior_active_key
    bool reset_occurred = false
    int n = math.min(array.size(intrabar_time), math.min(array.size(intrabar_open), math.min(array.size(intrabar_high), math.min(array.size(intrabar_low), math.min(array.size(intrabar_close), array.size(intrabar_volume))))))
    if n > 0
        for i = 0 to n - 1
            int intrabar_ts = array.get(intrabar_time, i)
            int key = ext_session_key(intrabar_ts, calendar_timezone, session_type, tokyo_start_h, tokyo_start_m, tokyo_end_h, tokyo_end_m, london_start_h, london_start_m, london_end_h, london_end_m, ny_start_h, ny_start_m, ny_end_h, ny_end_m)
            if not na(key)
                if na(active_key) or key != active_key
                    array.clear(stored_time)
                    array.clear(stored_open)
                    array.clear(stored_high)
                    array.clear(stored_low)
                    array.clear(stored_close)
                    array.clear(stored_volume)
                    active_key := key
                    reset_occurred := true
                int last_index = array.size(stored_time) - 1
                if last_index < 0 or intrabar_ts > array.get(stored_time, last_index)
                    array.push(stored_time, intrabar_ts)
                    array.push(stored_open, array.get(intrabar_open, i))
                    array.push(stored_high, array.get(intrabar_high, i))
                    array.push(stored_low, array.get(intrabar_low, i))
                    array.push(stored_close, array.get(intrabar_close, i))
                    array.push(stored_volume, array.get(intrabar_volume, i))
                else if intrabar_ts == array.get(stored_time, last_index)
                    array.set(stored_open, last_index, array.get(intrabar_open, i))
                    array.set(stored_high, last_index, array.get(intrabar_high, i))
                    array.set(stored_low, last_index, array.get(intrabar_low, i))
                    array.set(stored_close, last_index, array.get(intrabar_close, i))
                    array.set(stored_volume, last_index, array.get(intrabar_volume, i))
    [reset_occurred, active_key]
